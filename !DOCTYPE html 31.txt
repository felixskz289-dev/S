function checkGaps(){
  let ok=0,total=0;
  gapEl.querySelectorAll("input").forEach(inp=>{
    total++;
    const v=inp.value.trim(), a=inp.dataset.a;
    const gap=inp.parentElement;
    gap.classList.remove("ok","bad");
    if(v===a){ok++; gap.classList.add("ok");} else gap.classList.add("bad");
  });
  document.getElementById("gapresult").innerHTML =
    ok===total ? `<span style="color:var(--green)">Отлично! Всё верно: ${ok}/${total} 🎉</span>`
               : `<span style="color:var(--red)">Пока ${ok}/${total}. Проверь красные поля и попробуй ещё раз.</span>`;
}

let curBuild=0, built=[];
function shuffle(a){a=a.slice(); for(let i=a.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1)); [a[i],a[j]]=[a[j],a[i]];} return a;}
function renderBuild(){
  const t=builds[curBuild];
  document.getElementById("buildTask").innerHTML="Задание: <b>"+t.ru+"</b>";
  document.getElementById("slot").innerHTML="";
  document.getElementById("buildresult").innerHTML="";
  built=[];
  const bank=document.getElementById("bank"); bank.innerHTML="";
  const extra=["我","是","吗","，"];
  shuffle([...t.ans, ...shuffle(extra).slice(0,3)]).forEach(ch=>{
    const b=document.createElement("button"); b.className="chip"; b.textContent=ch;
    b.onclick=()=>{ built.push(ch); renderSlot(); };
    bank.appendChild(b);
  });
  renderSlot();
}
function renderSlot(){
  const s=document.getElementById("slot"); s.innerHTML="";
  built.forEach((ch,i)=>{
    const b=document.createElement("button"); b.className="chip"; b.textContent=ch;
    b.onclick=()=>{ built.splice(i,1); renderSlot(); };
    s.appendChild(b);
  });
}
function checkBuild(){
  const t=builds[curBuild];
  const good = built.join("")===t.ans.join("");
  document.getElementById("buildresult").innerHTML = good
    ? `<span style="color:var(--green)">Верно! ${t.ans.join("")} ✅</span>`
    : `<span style="color:var(--red)">Пока не так. Подсказка: правильный ответ — ${t.ans.join("")}</span>`;
}
function nextBuild(){ curBuild=(curBuild+1)%builds.length; renderBuild(); }
renderBuild();

        else{b.classList.add("wrong");}
        setTimeout(()=>{tstep++; renderTestQ();},900); };
      tb.appendChild(b);
    });
  } else {
    const wrap=document.createElement("div");
    wrap.innerHTML=`<div class="gap" style="min-width:90px"><input id="tgap" placeholder="……" style="width:90px"></div>
      <div style="margin-top:10px"><button class="checkbtn" id="tgo">Ответить</button></div>`;
    tb.appendChild(wrap);
    const inp=wrap.querySelector("#tgap"); inp.focus();
    const go=()=>{
      const ok=inp.value.trim()===q.ans; tstep++;
      wrap.querySelector(".gap").classList.add(ok?"ok":"bad");
      if(ok)tscore++;
      setTimeout(renderTestQ, ok?700:1600);
    };
    wrap.querySelector("#tgo").onclick=go;
    inp.onkeydown=e=>{if(e.key==="Enter")go();};
  }
}
